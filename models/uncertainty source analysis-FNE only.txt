### FNE is the only source of uncertainty

FVR_list_fne <- vector("list", nrow(pol_param))

set.seed(666)

for (i in seq_len(nrow(pol_param))) {
  r <- pol_param[i,]
  
  # no variance in body mass
  ms <- rep(r$mass_mean, N_FVR_DRAWS) 
  
  # no variance in FMR equation
  if (r$FMR_pred_a <= 0) stop("FMR_pred_a must be > 0") 
  FMR <- r$FMR_pred_a*ms^r$FMR_pred_b
  
  # no variance in FN
  FN  <- rep(r$FN_mean, N_FVR_DRAWS)

  # assuming FNE follows uniform dist
  FNE <- runif(N_FVR_DRAWS, r$FNE_min,  r$FNE_max)

  # no variance in NC and NV
  NV <- rep(r$NV_mean, N_FVR_DRAWS)
  NC <- rep(r$NC_mean, N_FVR_DRAWS)
  
  EperF <- NV * ul2ml * EC_sucorse * FNE * NC # energy provided by each flower in kJ/flower
  FVR   <- FMR * FN / EperF     # visits per day in flowers/day
  
  FVR_list_fne[[i]] <- data.table(
    species        = r$species,                 # aligned species name
    draw_fvr       = seq_len(N_FVR_DRAWS),      # local draw index
    FVR            = FVR,
    Service_days   = r$Service_days           # days stayed in North America
    # Breeding.Biome = r$Breeding.Biome,          # carried for grouping (if present)
    # Winter.Biome   = r$Winter.Biome,
    # Family         = r$Family,
    # Order          = r$Order,
    # specialization_lvl    = r$specialization_lvl,
    # Habitat          = r$Habitat
  )
}
FVR_dt_fne <- rbindlist(FVR_list_fne)

#----------------------------------------------------------------
# Calculating annual flower visits of each species
# FVR*Population abundance*Service days

K_pop_fne <- max(N_long$draw)

FVR_dt_fne[, draw := ((draw_fvr - 1L) %% K_pop_fne) + 1L]

# no variance in N
N1970_fne <- spyr_summ[year == 1970][, .(year, species, N_med)]
N2017_fne <- spyr_summ[year == 2017][, .(year, species, N_med)]

setkey(FVR_dt_fne, species, draw)
setkey(N1970_fne, species)
setkey(N2017_fne, species)

spyr_fne_1970 <- merge(
  N1970_fne,
  FVR_dt_fne[, .(species, draw, FVR, Service_days)],
  by = c("species"),
  allow.cartesian = TRUE
)

spyr_fne_2017 <- merge(
  N2017_fne,
  FVR_dt_fne[, .(species, draw, FVR, Service_days)],
  by = c("species"),
  allow.cartesian = TRUE
)

spyr_fne_1970[, annual.FV := FVR * Service_days * N_med] # visits per year
spyr_fne_2017[, annual.FV := FVR * Service_days * N_med] # visits per year


# -------------------------- Aggregations by continent ---------------------------------
# Continental level: sum across all species (year x draw)
cont_fne_1970 <- spyr_fne_1970[, .(annual.FV = sum(annual.FV)), by = .(year, draw)]
cont_fne_2017 <- spyr_fne_2017[, .(annual.FV = sum(annual.FV)), by = .(year, draw)]

cont_summ_fne_1970 <- cont_fne_1970[, {
  cAFV <- summarize_draws_90(annual.FV)
  out  <- setNames(cAFV, paste0("AFV_", names(cAFV)))
  out
}, by = character(0)]

cont_summ_fne_2017 <- cont_fne_2017[, {
  cAFV <- summarize_draws_90(annual.FV)
  out  <- setNames(cAFV, paste0("AFV_", names(cAFV)))
  out
}, by = character(0)]


# -------------------------- 5 to 95 percentile range -------------------

fne_range_1970 <- cont_summ_fne_1970$AFV_uci - cont_summ_fne_1970$AFV_lci
fne_range_2017 <- cont_summ_fne_2017$AFV_uci - cont_summ_fne_2017$AFV_lci


# ----------- Divide the percentile range of the individual errors by the total error ---------------------------------

fne_1970_error_prop <- fne_range_1970 / (cont_summ_90[year==1970]$AFV_uci-cont_summ_90[year==1970]$AFV_lci)
fne_2017_error_prop <- fne_range_2017 / (cont_summ_90[year==2017]$AFV_uci-cont_summ_90[year==2017]$AFV_lci)


# --------- Contribution to the uncertainty of cumulative change in continental-level AFV ---------------
cont_fne <- rbind(cont_fne_1970, cont_fne_2017)

cont_chg_summ_fne <- Change_trajectory_90(cont_fne,      value_col = "annual.FV", by_cols = character(0))

fne_range_chg <- cont_chg_summ_fne[year==2017]$uci - cont_chg_summ_fne[year==2017]$lci

fne_chg_prop <- fne_range_chg / (cont_chg_AFV_90[year==2017]$uci-cont_chg_AFV_90[year==2017]$lci)
