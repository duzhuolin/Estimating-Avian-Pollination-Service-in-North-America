### body mass is the only source of uncertainty

FVR_list_ms <- vector("list", nrow(pol_param))

set.seed(666)

for (i in seq_len(nrow(pol_param))) {
  r <- pol_param[i,]
  
  #assuming body mass follows truncated normal dist with SD being 15% of mean and lower bound being 1g
  ms <- TruncatedNormal::rtnorm(N_FVR_DRAWS, r$mass_mean, mass.sd*r$mass_mean, lb=1, ub=Inf) 
  
  # no variance in FMR equation
  if (r$FMR_pred_a <= 0) stop("FMR_pred_a must be > 0") 
  FMR <- r$FMR_pred_a*ms^r$FMR_pred_b
  
  # no variance in FN and FNE
  FN  <- rep((r$FN_min + r$FN_max) / 2, N_FVR_DRAWS)
  FNE <- rep((r$FNE_min + r$FNE_max) / 2, N_FVR_DRAWS)

  # no variance in NC and NV
  NV <- rep(exp(r$NV_mean_ln), N_FVR_DRAWS)
  NC <- rep(exp(r$NC_mean_ln), N_FVR_DRAWS)
  
  EperF <- NV * ul2ml * EC_sucorse * FNE * NC # energy provided by each flower in kJ/flower
  FVR   <- FMR * FN / EperF     # visits per day in flowers/day
  
  FVR_list_ms[[i]] <- data.table(
    species        = r$species,                 # aligned species name
    draw_fvr       = seq_len(N_FVR_DRAWS),      # local draw index
    FVR            = FVR,
    Service_days   = r$Service_days        # days stayed in North America
    # Breeding.Biome = r$Breeding.Biome,          # carried for grouping (if present)
    # Winter.Biome   = r$Winter.Biome,
    # Family         = r$Family,
    # Order          = r$Order,
    # specialization_lvl    = r$specialization_lvl,
    # Habitat          = r$Habitat
  )
}
FVR_dt_ms <- rbindlist(FVR_list_ms)

#----------------------------------------------------------------
# Calculating annual flower visits of each species
# FVR*Population abundance*Service days

K_pop_ms <- max(N_long$draw)

FVR_dt_ms[, draw := ((draw_fvr - 1L) %% K_pop_ms) + 1L]

# no variance in N
N1970_ms <- spyr_summ[year == 1970][, .(year, species, N_med)]
N2017_ms <- spyr_summ[year == 2017][, .(year, species, N_med)]

setkey(FVR_dt_ms, species, draw)
setkey(N1970_ms, species)
setkey(N2017_ms, species)

spyr_ms_1970 <- merge(
  N1970_ms,
  FVR_dt_ms[, .(species, draw, FVR, Service_days)],
  by = c("species"),
  allow.cartesian = TRUE
)

spyr_ms_2017 <- merge(
  N2017_ms,
  FVR_dt_ms[, .(species, draw, FVR, Service_days)],
  by = c("species"),
  allow.cartesian = TRUE
)

spyr_ms_1970[, annual.FV := FVR * Service_days * N_med] # visits per year
spyr_ms_2017[, annual.FV := FVR * Service_days * N_med] # visits per year


# -------------------------- Aggregations by continent ---------------------------------
# Continental level: sum across all species (year x draw)
cont_ms_1970 <- spyr_ms_1970[, .(annual.FV = sum(annual.FV)), by = .(year, draw)]
cont_ms_2017 <- spyr_ms_2017[, .(annual.FV = sum(annual.FV)), by = .(year, draw)]

cont_summ_ms_1970 <- cont_ms_1970[, {
  cAFV <- summarize_draws_90(annual.FV)
  out  <- setNames(cAFV, paste0("AFV_", names(cAFV)))
  out
}, by = year]

cont_summ_ms_2017 <- cont_ms_2017[, {
  cAFV <- summarize_draws_90(annual.FV)
  out  <- setNames(cAFV, paste0("AFV_", names(cAFV)))
  out
}, by = year]


# -------------------------- 5 to 95 percentile range -------------------

ms_range_1970 <- cont_summ_ms_1970$AFV_uci - cont_summ_ms_1970$AFV_lci
ms_range_2017 <- cont_summ_ms_2017$AFV_uci - cont_summ_ms_2017$AFV_lci


# ----------- Divide the percentile range of the individual errors by the total error ---------------------------------

ms_1970_error_prop <- ms_range_1970 / (cont_summ_90[year==1970]$AFV_uci-cont_summ_90[year==1970]$AFV_lci)
ms_2017_error_prop <- ms_range_2017 / (cont_summ_90[year==2017]$AFV_uci-cont_summ_90[year==2017]$AFV_lci)


# --------- Contribution to the uncertainty of cumulative change in continental-level AFV ---------------
cont_ms <- rbind(cont_ms_1970, cont_ms_2017)
