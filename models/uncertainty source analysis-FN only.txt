### FN is the only source of uncertainty

FVR_list_fn <- vector("list", nrow(pol_param))

set.seed(666)

for (i in seq_len(nrow(pol_param))) {
  r <- pol_param[i,]
  
  # no variance in body mass
  ms <- TruncatedNormal::qtnorm(0.5, 
                                r$mass_mean, 
                                mass.sd * r$mass_mean, 
                                lb = 1, ub = Inf)
  
  # no variance in FMR equation
  if (r$FMR_pred_a <= 0) stop("FMR_pred_a must be > 0") 
  FMR <- r$FMR_pred_a*ms^r$FMR_pred_b
  
  # assuming FN follows uniform dist
  FN  <- runif(N_FVR_DRAWS, r$FN_min,  r$FN_max)

  # no variance in FNE
  FNE <- rep((r$FNE_min + r$FNE_max) / 2, N_FVR_DRAWS)

  # no variance in NC and NV
  NV <- rep(exp(r$NV_mean_ln), N_FVR_DRAWS)
  NC <- rep(exp(r$NC_mean_ln), N_FVR_DRAWS)
  
  EperF <- NV * ul2ml * EC_sucorse * FNE * NC # energy provided by each flower in kJ/flower
  FVR   <- FMR * FN / EperF     # visits per day in flowers/day
  
  FVR_list_fn[[i]] <- data.table(
    species        = r$species,                 # aligned species name
    draw_fvr       = seq_len(N_FVR_DRAWS),      # local draw index
    FVR            = FVR,
    Service_days   = r$Service_days          # days stayed in North America
    # Breeding.Biome = r$Breeding.Biome,          # carried for grouping (if present)
    # Winter.Biome   = r$Winter.Biome,
    # Family         = r$Family,
    # Order          = r$Order,
    # specialization_lvl    = r$specialization_lvl,
    # Habitat          = r$Habitat
  )
}
FVR_dt_fn <- rbindlist(FVR_list_fn)

#----------------------------------------------------------------
# Calculating annual flower visits of each species
# FVR*Population abundance*Service days

K_pop_fn <- max(N_long$draw)

FVR_dt_fn[, draw := ((draw_fvr - 1L) %% K_pop_fn) + 1L]

# no variance in N
N1970_fn <- spyr_summ[year == 1970][, .(year, species, N_med)]
N2017_fn <- spyr_summ[year == 2017][, .(year, species, N_med)]

setkey(FVR_dt_fn, species, draw)
setkey(N1970_fn, species)
setkey(N2017_fn, species)

spyr_fn_1970 <- merge(
  N1970_fn,
  FVR_dt_fn[, .(species, draw, FVR, Service_days)],
  by = c("species"),
  allow.cartesian = TRUE
)

spyr_fn_2017 <- merge(
  N2017_fn,
  FVR_dt_fn[, .(species, draw, FVR, Service_days)],
  by = c("species"),
  allow.cartesian = TRUE
)

spyr_fn_1970[, annual.FV := FVR * Service_days * N_med] # visits per year
spyr_fn_2017[, annual.FV := FVR * Service_days * N_med] # visits per year


# -------------------------- Aggregations by continent ---------------------------------
# Continental level: sum across all species (year x draw)
cont_fn_1970 <- spyr_fn_1970[, .(annual.FV = sum(annual.FV)), by = .(year, draw)]
cont_fn_2017 <- spyr_fn_2017[, .(annual.FV = sum(annual.FV)), by = .(year, draw)]

cont_summ_fn_1970 <- cont_fn_1970[, {
  cAFV <- summarize_draws_90(annual.FV)
  out  <- setNames(cAFV, paste0("AFV_", names(cAFV)))
  out
}, by = character(0)]

cont_summ_fn_2017 <- cont_fn_2017[, {
  cAFV <- summarize_draws_90(annual.FV)
  out  <- setNames(cAFV, paste0("AFV_", names(cAFV)))
  out
}, by = character(0)]


# -------------------------- 5 to 95 percentile range -------------------

fn_range_1970 <- cont_summ_fn_1970$AFV_uci - cont_summ_fn_1970$AFV_lci
fn_range_2017 <- cont_summ_fn_2017$AFV_uci - cont_summ_fn_2017$AFV_lci


# ----------- Divide the percentile range of the individual errors by the total error ---------------------------------

fn_1970_error_prop <- fn_range_1970 / (cont_summ_90[year==1970]$AFV_uci-cont_summ_90[year==1970]$AFV_lci)
fn_2017_error_prop <- fn_range_2017 / (cont_summ_90[year==2017]$AFV_uci-cont_summ_90[year==2017]$AFV_lci)


# --------- Contribution to the uncertainty of cumulative change in continental-level AFV ---------------
cont_fn <- rbind(cont_fn_1970, cont_fn_2017)

cont_chg_summ_fn <- Change_trajectory_90(cont_fn,      value_col = "annual.FV", by_cols = character(0))

fn_range_chg <- cont_chg_summ_fn[year==2017]$uci - cont_chg_summ_fn[year==2017]$lci

fn_chg_prop <- fn_range_chg / (cont_chg_AFV_90[year==2017]$uci-cont_chg_AFV_90[year==2017]$lci)
